cmake_minimum_required(VERSION 3.8)
project(mobile_dual_nero)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(std_srvs REQUIRED)
add_library(bottle_grasp SHARED src/bottle_grasp_plugin.cpp)
target_include_directories(bottle_grasp PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(bottle_grasp ${GAZEBO_LIBRARIES})
ament_target_dependencies(bottle_grasp gazebo_ros std_srvs)
install(TARGETS bottle_grasp LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
add_executable(add_ground src/add_ground.cpp)
ament_target_dependencies(add_ground rclcpp moveit_ros_planning)
install(TARGETS add_ground DESTINATION lib/${PROJECT_NAME})
add_executable(official_command_bridge src/official_command_bridge.cpp)
target_compile_features(official_command_bridge PRIVATE cxx_std_17)
ament_target_dependencies(official_command_bridge rclcpp moveit_ros_planning_interface moveit_ros_planning tf2_ros geometry_msgs std_msgs sensor_msgs visualization_msgs)
install(TARGETS official_command_bridge DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/verify_assembly scripts/sync_bottle DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(composition tests/test_composition.py)
  ament_add_pytest_test(environment tests/test_environment.py)
  ament_add_pytest_test(bottle_sync tests/test_bottle_sync.py)
endif()
ament_package()
