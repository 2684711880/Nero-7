cmake_minimum_required(VERSION 3.8)
project(mobile_dual_nero)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_ros_planning REQUIRED)
ament_python_install_package(${PROJECT_NAME})
add_executable(add_ground src/add_ground.cpp)
ament_target_dependencies(add_ground rclcpp moveit_ros_planning)
install(TARGETS add_ground DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/verify_assembly DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(composition tests/test_composition.py)
endif()
ament_package()
