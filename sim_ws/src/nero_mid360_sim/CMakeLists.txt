cmake_minimum_required(VERSION 3.8)
project(nero_mid360_sim)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
find_package(rclcpp REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(nero_mid360 SHARED src/mid360_plugin.cpp)
target_compile_features(nero_mid360 PRIVATE cxx_std_17)
target_include_directories(nero_mid360 PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(nero_mid360 ${GAZEBO_LIBRARIES})
ament_target_dependencies(nero_mid360 gazebo_ros rclcpp livox_ros_driver2 sensor_msgs std_msgs)
install(TARGETS nero_mid360 LIBRARY DESTINATION lib)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/mapping_robot_pose scripts/fastlio_output_adapter scripts/body_self_filter DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(body_self_filter tests)
endif()
ament_package()
