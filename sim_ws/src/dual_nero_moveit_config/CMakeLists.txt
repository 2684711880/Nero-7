cmake_minimum_required(VERSION 3.8)
project(dual_nero_moveit_config)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(moveit_ros_planning REQUIRED)
add_executable(verify_dual src/verify_dual.cpp)
ament_target_dependencies(verify_dual rclcpp moveit_ros_planning_interface moveit_ros_planning)
add_executable(add_ground src/add_ground.cpp)
ament_target_dependencies(add_ground rclcpp moveit_ros_planning)
install(TARGETS add_ground verify_dual DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config urdf meshes launch worlds DESTINATION share/${PROJECT_NAME})
install(FILES README.md ASSET_SOURCE.md UPSTREAM_LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
