cmake_minimum_required(VERSION 3.8)
project(nero_official_commander_cpp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(gazebo_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(std_srvs REQUIRED)
add_executable(pick_place src/pick_place.cpp)
target_include_directories(pick_place PRIVATE include)
target_compile_features(pick_place PRIVATE cxx_std_17)
target_compile_options(pick_place PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(pick_place rclcpp geometry_msgs tf2_ros std_msgs sensor_msgs gazebo_msgs moveit_msgs shape_msgs std_srvs)
install(TARGETS pick_place DESTINATION lib/${PROJECT_NAME})
add_executable(commander src/commander.cpp)
target_include_directories(commander PRIVATE include)
target_compile_features(commander PRIVATE cxx_std_17)
target_compile_options(commander PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(commander rclcpp geometry_msgs tf2_ros std_msgs sensor_msgs)
install(TARGETS commander DESTINATION lib/${PROJECT_NAME})
add_executable(commander_reference src/commander_reference.cpp)
target_compile_features(commander_reference PRIVATE cxx_std_17)
ament_target_dependencies(commander_reference rclcpp geometry_msgs tf2_ros std_msgs)
install(TARGETS commander_reference DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(api_smoke test/api_smoke.cpp)
  target_include_directories(api_smoke PRIVATE include)
  target_compile_features(api_smoke PRIVATE cxx_std_17)
  ament_target_dependencies(api_smoke rclcpp geometry_msgs tf2_ros std_msgs sensor_msgs)
  install(TARGETS api_smoke DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
