cmake_minimum_required(VERSION 3.8)
project(nero_official_commander_cpp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)

find_package(shape_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(moveit_core REQUIRED)
add_executable(straight_line src/straight_line.cpp)
target_compile_features(straight_line PRIVATE cxx_std_17)
target_compile_options(straight_line PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(straight_line rclcpp sensor_msgs std_msgs moveit_msgs moveit_ros_planning_interface moveit_core)
install(TARGETS straight_line DESTINATION lib/${PROJECT_NAME})
